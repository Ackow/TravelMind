> 目标：亲手建立 TravelMind 的结构化领域底座，并用固定东京旅行数据证明对象可以校验、加载和序列化。  
> 本阶段不实现规划算法、约束引擎、数据库、API 路由、LLM、LangGraph 或真实第三方服务。

## 3. 本阶段的依赖方向

```text
scripts / tests / future application layer
                  ↓
              domain
                  ↑
            fixture loader
```

准确地说，fixture loader 可以导入 domain；domain 不能反过来导入 fixture loader。

`backend/app/domain/` 中禁止导入：

- FastAPI；
- LangGraph；
- SQLAlchemy；
- 第三方天气或地图 SDK；
- `app.api`；
- `app.persistence`。

领域模型只描述业务数据和跨入口都成立的不变量。

## 5. 第一步：实现通用值对象

创建 `backend/app/domain/common.py`：

```python
from datetime import date, datetime
from enum import StrEnum

from pydantic import BaseModel, ConfigDict, Field, model_validator


class DomainModel(BaseModel):
    """所有领域对象的共同配置。"""

    model_config = ConfigDict(
        extra="forbid",
        str_strip_whitespace=True,
        validate_assignment=True,
    )


class Money(DomainModel):
    amount: int = Field(ge=0, description="最小货币单位整数")
    currency: str = Field(pattern=r"^[A-Z]{3}$")


class GeoPoint(DomainModel):
    latitude: float = Field(ge=-90, le=90)
    longitude: float = Field(ge=-180, le=180)


class DateRange(DomainModel):
    start_date: date
    end_date: date

    @model_validator(mode="after")
    def validate_range(self) -> "DateRange":
        if self.end_date < self.start_date:
            raise ValueError("end_date must not be before start_date")
        return self

    @property
    def day_count(self) -> int:
        return (self.end_date - self.start_date).days + 1


class DataQuality(StrEnum):
    VERIFIED = "verified"
    ESTIMATED = "estimated"
    INCOMPLETE = "incomplete"
    STALE = "stale"
    MOCK = "mock"


class SourceRef(DomainModel):
    provider: str = Field(min_length=1, max_length=50)
    source_id: str | None = None
    source_url: str | None = None
    fetched_at: datetime
    expires_at: datetime | None = None
    data_quality: DataQuality

    @model_validator(mode="after")
    def validate_timestamps(self) -> "SourceRef":
        if self.fetched_at.tzinfo is None:
            raise ValueError("fetched_at must be timezone-aware")
        if self.expires_at is not None:
            if self.expires_at.tzinfo is None:
                raise ValueError("expires_at must be timezone-aware")
            if self.expires_at <= self.fetched_at:
                raise ValueError("expires_at must be after fetched_at")
        return self
```

### 你需要理解的点

#### 为什么 `Money.amount` 不用 `float`

二进制浮点数不能精确表示许多十进制金额。例如 0.1 + 0.2 不一定精确等于 0.3。使用最小货币单位整数可以避免累计误差：

```text
CNY 100.00 元 → amount = 10000
JPY 1000 日元 → amount = 1000
```

注意：日元通常没有常用小数单位，人民币有分。显示层负责根据币种格式化。

#### 为什么 `extra="forbid"`

如果 JSON 写错 `curreny`，默认忽略它会让错误很难发现。禁止未知字段可以尽早暴露契约漂移和拼写错误。

#### 为什么要求 aware datetime

`2026-10-02 09:00` 无法说明这是东京时间、北京时间还是 UTC。所有具体时间点必须带时区偏移；旅行日期则继续用当地 `date`。

## 6. 第二步：实现旅行需求对象

创建 `backend/app/domain/trip.py`：

```python
from enum import StrEnum
from zoneinfo import ZoneInfo, ZoneInfoNotFoundError

from pydantic import Field, field_validator, model_validator

from app.domain.common import DateRange, DomainModel, Money


class Pace(StrEnum):
    RELAXED = "relaxed"
    BALANCED = "balanced"
    PACKED = "packed"


class TransportMode(StrEnum):
    WALKING = "walking"
    PUBLIC_TRANSIT = "public_transit"
    TAXI = "taxi"
    DRIVING = "driving"
    CYCLING = "cycling"
    MIXED = "mixed"


class DietaryPreference(StrEnum):
    VEGETARIAN = "vegetarian"
    VEGAN = "vegan"
    HALAL = "halal"
    KOSHER = "kosher"
    GLUTEN_FREE = "gluten_free"
    NO_PORK = "no_pork"
    NO_BEEF = "no_beef"
    SEAFOOD_FREE = "seafood_free"
    NUT_FREE = "nut_free"


class WeightedPreference(DomainModel):
    value: str = Field(min_length=1, max_length=50)
    weight: float = Field(gt=0, le=1)


class TripPreferences(DomainModel):
    interests: list[WeightedPreference] = Field(default_factory=list)
    avoid: list[str] = Field(default_factory=list)
    dietary: list[DietaryPreference] = Field(default_factory=list)
    transport_modes: list[TransportMode] = Field(
        default_factory=lambda: [
            TransportMode.PUBLIC_TRANSIT,
            TransportMode.WALKING,
        ]
    )
    accommodation_notes: str | None = Field(default=None, max_length=500)
    pace: Pace = Pace.BALANCED
    must_visit_place_names: list[str] = Field(default_factory=list)

    @model_validator(mode="after")
    def validate_unique_interests(self) -> "TripPreferences":
        values = [item.value.casefold() for item in self.interests]
        if len(values) != len(set(values)):
            raise ValueError("interest values must be unique")
        return self


class TripConstraints(DomainModel):
    total_budget: Money
    budget_is_hard_limit: bool = True
    daily_start_time: str = Field(default="09:00", pattern=r"^(?:[01]\d|2[0-3]):[0-5]\d$")
    daily_end_time: str = Field(default="21:00", pattern=r"^(?:[01]\d|2[0-3]):[0-5]\d$")
    max_walking_meters_per_day: int | None = Field(default=12000, ge=1000, le=50000)
    max_activities_per_day: int = Field(default=5, ge=1, le=10)
    minimum_transfer_buffer_minutes: int = Field(default=10, ge=0, le=60)
    rest_minutes_per_day: int = Field(default=60, ge=0, le=240)
    required_place_names: list[str] = Field(default_factory=list)
    excluded_place_names: list[str] = Field(default_factory=list)
    accessible_only: bool = False

    @model_validator(mode="after")
    def validate_constraint_consistency(self) -> "TripConstraints":
        if self.daily_end_time <= self.daily_start_time:
            raise ValueError("daily_end_time must be after daily_start_time")

        required = {name.casefold() for name in self.required_place_names}
        excluded = {name.casefold() for name in self.excluded_place_names}
        overlap = required & excluded
        if overlap:
            raise ValueError("a place cannot be both required and excluded")
        return self


class TripRequest(DomainModel):
    origin: str = Field(min_length=1, max_length=100)
    destination: str = Field(min_length=1, max_length=100)
    destination_timezone: str
    date_range: DateRange
    travelers: int = Field(ge=1, le=6)
    preferences: TripPreferences
    constraints: TripConstraints
    locale: str = Field(default="zh-CN", pattern=r"^[a-z]{2,3}(?:-[A-Z]{2})?$")
    display_currency: str = Field(default="CNY", pattern=r"^[A-Z]{3}$")
    notes: str | None = Field(default=None, max_length=1000)

    @field_validator("destination_timezone")
    @classmethod
    def validate_timezone(cls, value: str) -> str:
        try:
            ZoneInfo(value)
        except ZoneInfoNotFoundError as exc:
            raise ValueError("destination_timezone must be a valid IANA timezone") from exc
        return value

    @model_validator(mode="after")
    def validate_mvp_scope(self) -> "TripRequest":
        if not 3 <= self.date_range.day_count <= 7:
            raise ValueError("MVP trips must contain between 3 and 7 days")
        if self.display_currency != self.constraints.total_budget.currency:
            raise ValueError("display currency must match budget currency in stage 1")
        return self
```

### 为什么偏好与约束分开

偏好可以用于评分和取舍：

```text
喜欢动漫、倾向公共交通、节奏轻松
```

约束决定方案是否可行：

```text
总预算、每日结束时间、必去/禁去地点
```

如果二者混成一个 `list[str]`，阶段 2 的规则引擎就无法知道哪些条件必须满足。

### 为什么 `required_place_names` 暂时使用名称

创建旅行时还没有调用 POI 工具，尚未得到稳定 Place ID。阶段 1 保存用户原始名称；后续 research 阶段解析成候选地点，再保留名称与 ID 的映射证据。

## 7. 第三步：实现天气、POI 与路线事实对象

创建 `backend/app/domain/research.py`：

```python
from datetime import date, time
from enum import StrEnum

from pydantic import Field, model_validator

from app.domain.common import DomainModel, GeoPoint, Money, SourceRef
from app.domain.trip import TransportMode


class WeatherCondition(StrEnum):
    CLEAR = "clear"
    PARTLY_CLOUDY = "partly_cloudy"
    CLOUDY = "cloudy"
    RAIN = "rain"
    STORM = "storm"
    SNOW = "snow"
    FOG = "fog"
    UNKNOWN = "unknown"


class OutdoorSuitability(StrEnum):
    GOOD = "good"
    ACCEPTABLE = "acceptable"
    POOR = "poor"
    UNKNOWN = "unknown"


class IndoorOutdoor(StrEnum):
    INDOOR = "indoor"
    OUTDOOR = "outdoor"
    MIXED = "mixed"
    UNKNOWN = "unknown"


class PlaceCategory(StrEnum):
    ATTRACTION = "attraction"
    MUSEUM = "museum"
    PARK = "park"
    ANIME = "anime"
    FOOD = "food"
    RESTAURANT = "restaurant"
    CAFE = "cafe"
    SHOPPING = "shopping"
    NEIGHBORHOOD = "neighborhood"
    TEMPLE = "temple"
    SHRINE = "shrine"
    VIEWPOINT = "viewpoint"
    ENTERTAINMENT = "entertainment"


class WeatherDay(DomainModel):
    date: date
    condition: WeatherCondition
    temperature_min_c: float | None = None
    temperature_max_c: float | None = None
    rain_probability: float | None = Field(default=None, ge=0, le=1)
    precipitation_mm: float | None = Field(default=None, ge=0)
    sunrise_time: time | None = None
    sunset_time: time | None = None
    outdoor_suitability: OutdoorSuitability
    source: SourceRef

    @model_validator(mode="after")
    def validate_temperature_range(self) -> "WeatherDay":
        if (
            self.temperature_min_c is not None
            and self.temperature_max_c is not None
            and self.temperature_min_c > self.temperature_max_c
        ):
            raise ValueError("temperature_min_c must not exceed temperature_max_c")
        return self


class OpeningPeriod(DomainModel):
    day_of_week: int = Field(ge=1, le=7)
    open_time: time | None = None
    close_time: time | None = None
    closed: bool

    @model_validator(mode="after")
    def validate_times(self) -> "OpeningPeriod":
        if self.closed:
            if self.open_time is not None or self.close_time is not None:
                raise ValueError("closed period must not contain opening times")
            return self

        if self.open_time is None or self.close_time is None:
            raise ValueError("open period must contain open_time and close_time")
        if self.close_time <= self.open_time:
            raise ValueError("stage 1 does not support opening periods crossing midnight")
        return self


class SpecialOpeningPeriod(DomainModel):
    date: date
    open_time: time | None = None
    close_time: time | None = None
    closed: bool
    note: str | None = Field(default=None, max_length=200)

    @model_validator(mode="after")
    def validate_times(self) -> "SpecialOpeningPeriod":
        if self.closed:
            if self.open_time is not None or self.close_time is not None:
                raise ValueError("closed special period must not contain times")
            return self
        if self.open_time is None or self.close_time is None:
            raise ValueError("open special period must contain times")
        if self.close_time <= self.open_time:
            raise ValueError("special opening period must end after it starts")
        return self


class Place(DomainModel):
    id: str = Field(min_length=1, max_length=100)
    name: str = Field(min_length=1, max_length=200)
    localized_name: str | None = Field(default=None, max_length=200)
    categories: list[PlaceCategory] = Field(min_length=1)
    address: str | None = Field(default=None, max_length=500)
    location: GeoPoint
    rating: float | None = Field(default=None, ge=0, le=5)
    rating_count: int | None = Field(default=None, ge=0)
    estimated_visit_minutes: int = Field(ge=15, le=720)
    indoor_outdoor: IndoorOutdoor
    opening_periods: list[OpeningPeriod] = Field(default_factory=list)
    special_opening_periods: list[SpecialOpeningPeriod] = Field(default_factory=list)
    admission: Money | None = None
    tags: list[str] = Field(default_factory=list)
    reservation_required: bool | None = None
    accessible: bool | None = None
    website_url: str | None = None
    source: SourceRef


class RoutePoint(DomainModel):
    place_id: str
    location: GeoPoint


class RouteMatrixStatus(StrEnum):
    OK = "ok"
    UNREACHABLE = "unreachable"
    UNKNOWN = "unknown"


class RouteMatrixCell(DomainModel):
    origin_place_id: str
    destination_place_id: str
    mode: TransportMode
    status: RouteMatrixStatus
    duration_minutes: int | None = Field(default=None, ge=0)
    distance_meters: int | None = Field(default=None, ge=0)
    walking_meters: int | None = Field(default=None, ge=0)
    cost: Money | None = None

    @model_validator(mode="after")
    def validate_metrics(self) -> "RouteMatrixCell":
        metrics = (
            self.duration_minutes,
            self.distance_meters,
            self.walking_meters,
        )
        if self.status == RouteMatrixStatus.OK and any(value is None for value in metrics):
            raise ValueError("ok route cell must contain duration, distance and walking distance")
        return self


class RouteMatrix(DomainModel):
    cells: list[RouteMatrixCell]
    source: SourceRef

    @model_validator(mode="after")
    def validate_unique_pairs(self) -> "RouteMatrix":
        keys = [
            (cell.origin_place_id, cell.destination_place_id, cell.mode)
            for cell in self.cells
        ]
        if len(keys) != len(set(keys)):
            raise ValueError("route matrix cells must be unique by origin, destination and mode")
        return self
```

### 一个重要建模决定

阶段 1 的固定路线数据使用 `RouteMatrixCell`，而不是最终计划里的 `RouteLeg`：

- `RouteMatrixCell` 是研究阶段的候选地点间估算事实；
- `RouteLeg` 是规划完成后带具体出发/到达时间的实际行程路段。

不要把“候选事实”和“已排入计划的结果”混成同一个对象。

## 8. 第四步：实现空白行程对象

阶段 1 只生成空白骨架，不安排地点。创建 `backend/app/domain/itinerary.py`：

```python
from datetime import date, datetime
from uuid import UUID

from pydantic import Field, model_validator

from app.domain.common import DateRange, DomainModel, Money
from app.domain.research import WeatherDay


class DayStatistics(DomainModel):
    activity_count: int = Field(default=0, ge=0)
    walking_meters: int = Field(default=0, ge=0)
    transfer_minutes: int = Field(default=0, ge=0)
    planned_minutes: int = Field(default=0, ge=0)
    estimated_cost: Money


class DayPlan(DomainModel):
    date: date
    day_number: int = Field(ge=1)
    theme: str = Field(default="待规划", min_length=1, max_length=100)
    weather: WeatherDay | None = None
    activities: list[dict] = Field(default_factory=list)
    route_legs: list[dict] = Field(default_factory=list)
    statistics: DayStatistics
    warnings: list[str] = Field(default_factory=list)


class BudgetSummary(DomainModel):
    limit: Money
    planned_total: Money
    remaining_amount: int
    currency: str = Field(pattern=r"^[A-Z]{3}$")
    within_budget: bool

    @model_validator(mode="after")
    def validate_totals(self) -> "BudgetSummary":
        if self.limit.currency != self.currency:
            raise ValueError("limit currency must match summary currency")
        if self.planned_total.currency != self.currency:
            raise ValueError("planned total currency must match summary currency")

        expected_remaining = self.limit.amount - self.planned_total.amount
        if self.remaining_amount != expected_remaining:
            raise ValueError("remaining_amount does not match budget totals")
        if self.within_budget != (expected_remaining >= 0):
            raise ValueError("within_budget does not match remaining amount")
        return self


class Itinerary(DomainModel):
    trip_id: UUID
    title: str = Field(min_length=1, max_length=200)
    destination: str = Field(min_length=1, max_length=100)
    timezone: str
    date_range: DateRange
    days: list[DayPlan]
    budget: BudgetSummary
    general_notes: list[str] = Field(default_factory=list)
    generated_at: datetime

    @model_validator(mode="after")
    def validate_days(self) -> "Itinerary":
        if self.generated_at.tzinfo is None:
            raise ValueError("generated_at must be timezone-aware")
        if len(self.days) != self.date_range.day_count:
            raise ValueError("itinerary must contain one day plan per trip date")

        expected_dates = [
            self.date_range.start_date.fromordinal(
                self.date_range.start_date.toordinal() + offset
            )
            for offset in range(self.date_range.day_count)
        ]
        actual_dates = [day.date for day in self.days]
        if actual_dates != expected_dates:
            raise ValueError("day plans must be ordered and cover the complete date range")
        return self
```

```python
from pathlib import Path

from pydantic import TypeAdapter

from app.domain.research import Place, RouteMatrix, WeatherDay

FIXTURE_ROOT = Path(__file__).resolve().parent
```

如果想清楚掌握每个环节，建议分四次提交：


## 16. 本阶段复盘问题

完成后不要立刻进入阶段 2，先用自己的话回答：

1. 为什么 API 请求 DTO、领域模型和数据库模型以后可能不是同一个类？
2. 为什么预算用整数而不是 float？
3. `null`、缺失字段和数值 0 各自代表什么？
4. 为什么天气和 POI 必须携带来源及获取时间？
5. 为什么路线矩阵事实不等于最终行程中的 RouteLeg？
6. 为什么用户输入的必去名称不能立即当作真实 Place？
7. 哪些不变量适合放 Pydantic，哪些应留给阶段 2 的 Constraint Engine？

最后一个问题尤其重要：

- “结束日期不能早于开始日期”是对象自身永远成立的不变量，放在模型中；
- “某项活动是否发生在闭馆之后”需要同时读取行程和 POI 事实，放在 Constraint Engine 中。

能清晰区分这两类校验，就真正掌握了阶段 1 的核心。
